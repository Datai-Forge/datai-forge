1import logging 

2import os 

3from pyspark.sql import functions as F 

4from src.common.spark_session_manager import get_spark_session 

5from src.config import SILVER_PATH, GOLD_PATH 

6 

7# Configuration du logging 

8logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s') 

9logger = logging.getLogger(__name__)

11def run_gold_securite_bi_pipeline(): 

12 """ 

13 Transforme les données Silver de sécurité en schéma en étoile pour la BI (Lyon). 

14 - dim_indicateurs_securite : Référentiel des types de crimes (15 indicateurs) 

15 - dim_geographie_lyon : Référentiel des arrondissements (69381-69389) 

16 - fact_securite : Métriques de délinquance par année/arrondissement 

17 - fact_demographie_annuelle : Population et logements par année/arrondissement (évite le double-comptage) 

18 """ 

19 spark = get_spark_session("Gold_Securite_BI") 

20 logger.info("Démarrage de la pipeline Gold Securite BI...") 

21 

22 # Chargement des données Silver 

23 silver_securite_path = os.path.join(SILVER_PATH, "securite", "lyon_securite") 

24 

25 if not os.path.exists(silver_securite_path): 

26 logger.error(f"Données Silver introuvables : {silver_securite_path}") 

27 return 

28 

29 df_silver = spark.read.parquet(silver_securite_path) 

30 

31 # Ajout de l'année pour simplifier les tables de faits 

32 df_silver = df_silver.withColumn("annee", F.year("date_reference")) 

33 

34 # Dimension Indicateurs (Les 15 types de crimes) 

35 logger.info("Extraction de dim_indicateurs_securite...") 

36 dim_indicateurs = df_silver.select( 

37 F.col("indicateur").alias("id_indicateur"), 

38 "unite_de_compte" 

39 ).distinct() 

40 

41 # Dimension Géographie (Arrondissements de Lyon) 

42 logger.info("Extraction de dim_geographie_lyon...") 

43 dim_geographie = df_silver.select( 

44 F.col("code_geo").alias("code_arrondissement") 

45 ).distinct().withColumn("nom_arrondissement", 

46 F.when(F.col("code_arrondissement") == "69381", "1er Arrondissement") 

47 .when(F.col("code_arrondissement") == "69382", "2ème Arrondissement") 

48 .when(F.col("code_arrondissement") == "69383", "3ème Arrondissement") 

49 .when(F.col("code_arrondissement") == "69384", "4ème Arrondissement") 

50 .when(F.col("code_arrondissement") == "69385", "5ème Arrondissement") 

51 .when(F.col("code_arrondissement") == "69386", "6ème Arrondissement") 

52 .when(F.col("code_arrondissement") == "69387", "7ème Arrondissement") 

53 .when(F.col("code_arrondissement") == "69388", "8ème Arrondissement") 

54 .when(F.col("code_arrondissement") == "69389", "9ème Arrondissement") 

55 .otherwise("Inconnu") 

56 ) 

57 

58 ## Deux tables des faits car on veut éviter de sommer la population 15 fois en BI (une pour chaque indicateur) : 

59 # Table de faits : Sécurité (Nombre et Taux) 

60 logger.info("Extraction de fact_securite...") 

61 fact_securite = df_silver.select( 

62 F.col("code_geo").alias("code_arrondissement"), 

63 F.col("indicateur").alias("id_indicateur"), 

64 "annee", 

65 "nombre", 

66 "taux_pour_1000" 

67 ) 

68 

69 # Table de faits : Démographie (Socio-éco contextuel) 

70 # Règle Métier : On isole la population pour éviter de la sommer 15 fois en BI 

71 logger.info("Extraction de fact_demographie_annuelle...") 

72 fact_demographie = df_silver.select( 

73 F.col("code_geo").alias("code_arrondissement"), 

74 "annee", 

75 "population", 

76 "logements" 

77 ).distinct() 

78 

79 # Définition du chemin de sortie 

80 output_base_path = os.path.join(GOLD_PATH, "securite", "bi") 

81 logger.info(f"Sauvegarde des tables Gold Securite BI dans {output_base_path}...") 

82 

83 # Sauvegarde au format Parquet 

84 dim_indicateurs.write.mode("overwrite").parquet(os.path.join(output_base_path, "dim_indicateurs_securite")) 

85 dim_geographie.write.mode("overwrite").parquet(os.path.join(output_base_path, "dim_geographie_lyon")) 

86 fact_securite.write.mode("overwrite").parquet(os.path.join(output_base_path, "fact_securite")) 

87 fact_demographie.write.mode("overwrite").parquet(os.path.join(output_base_path, "fact_demographie_annuelle")) 

88 

89 # Audit final pour vérification 

90 nb_indicateurs = dim_indicateurs.count() 

91 nb_arrondissements = dim_geographie.count() 

92 nb_lignes_faits = fact_securite.count() 

93 

94 logger.info(f"Audit Gold : {nb_indicateurs} indicateurs sur {nb_arrondissements} arrondissements.") 

95 logger.info(f"Total de {nb_lignes_faits} mesures de sécurité générées.") 

96 

97 logger.info("Pipeline Gold Securite BI terminée avec succès !") 

98 

99if __name__ == "__main__": 

100 run_gold_securite_bi_pipeline()
