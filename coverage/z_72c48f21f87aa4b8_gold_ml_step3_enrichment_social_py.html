15def run_gold_ml_step3_enrichment_social(): 

16 """ 

17 Étape 3 : Enrichissement final de l'OBT avec les données Socio-Économiques. 

18 - Agrégation massive des données Insee Silver par Arrondissement. 

19 - Calcul des features sociales (Ratios et Deltas). 

20 - Jointure finale pour obtenir l'OBT ML complète. 

21 """ 

22 spark = get_spark_session("Gold_ML_Step3_Social") 

23 logger.info("Démarrage de l'étape 3 : Enrichissement Socio-Économique...") 

24 

25 # Chargement de l'OBT existante (Elections + Sécurité) 

26 obt_step2_path = os.path.join(GOLD_PATH, "ml", "obt_step2_security") 

27 df_obt = spark.read.parquet(obt_step2_path) 

28 

29 # Chargement et Fusion des données Silver Social (2017, 2019, 2021) 

30 base_social_path = os.path.join(SILVER_PATH, "niveau_vie_pauvrete") 

31 

32 dfs_social = [] 

33 for year in [2017, 2019, 2021]: 

34 path = os.path.join(base_social_path, f"niveau_vie_pauvrete_{year}") 

35 if not os.path.exists(path): 

36 logger.warning(f"Dossier {path} introuvable, skip.") 

37 continue 

38 df_year = spark.read.parquet(path).withColumn("annee", F.lit(year)) 

39 

40 # On utilise la colonne 'arrondissement' (ex: 69381) pour la jointure 

41 df_year = df_year.withColumn("id_arr_social", F.col("arrondissement")) 

42 dfs_social.append(df_year) 

43 

44 if not dfs_social: 

45 logger.error("Aucune donnée sociale trouvée !") 

46 return 

47 

48 df_social_all = dfs_social[0] 

49 for df in dfs_social[1:]: 

50 df_social_all = df_social_all.unionByName(df, allowMissingColumns=True) 

51 

52 # agrégation par Arrondissement et par Année 

53 numeric_types = (FloatType, DoubleType, LongType, IntegerType) 

54 cols_to_sum = [ 

55 field.name 

56 for field in df_social_all.schema.fields 

57 if isinstance(field.dataType, numeric_types) 

58 and field.name not in ["annee", "id_arr_social"] 

59 ] 

60 

61 logger.info(f"Agrégation de {len(cols_to_sum)} indicateurs sociaux...") 

62 df_agg = df_social_all.groupBy("id_arr_social", "annee").agg( 

63 *[F.sum(c).alias(c) for c in cols_to_sum] 

64 ) 

65 

66 # Feature Engineering : Calcul des Ratios par Arrondissement 

67 df_features = ( 

68 df_agg.withColumn( 

69 "feat_social_taux_pauvrete", 

70 F.round((F.col("nb_menages_pauvres") / F.col("nb_menages")) * 100, 2), 

71 ) 

72 .withColumn( 

73 "feat_social_revenu_moyen", 

74 F.round( 

75 F.col("somme_niveaux_de_vie_winsorises_des_individus") / F.col("nb_individus"), 2 

76 ), 

77 ) 

78 .withColumn( 

79 "feat_social_pct_proprietaires", 

80 F.round((F.col("nb_menages_propriétaires") / F.col("nb_menages")) * 100, 2), 

81 ) 

82 .withColumn( 

83 "feat_social_pct_logements_sociaux", 

84 F.round((F.col("nb_logements_sociaux") / F.col("nb_menages")) * 100, 2), 

85 ) 

86 ) 

87 

88 # Calcul des Deltas (2021 vs 2017) 

89 df_2017 = df_features.filter(F.col("annee") == 2017).select( 

90 F.col("id_arr_social"), 

91 F.col("feat_social_revenu_moyen").alias("rev17"), 

92 F.col("feat_social_taux_pauvrete").alias("pauv17"), 

93 ) 

94 

95 df_2021 = df_features.filter(F.col("annee") == 2021).select( 

96 "id_arr_social", 

97 "feat_social_revenu_moyen", 

98 "feat_social_taux_pauvrete", 

99 "feat_social_pct_proprietaires", 

100 "feat_social_pct_logements_sociaux", 

101 ) 

102 

103 df_social_final = ( 

104 df_2021.join(df_2017, "id_arr_social", "inner") 

105 .withColumn( 

106 "feat_social_delta_revenu_5ans", 

107 F.round( 

108 ((F.col("feat_social_revenu_moyen") - F.col("rev17")) / F.col("rev17")) * 100, 2 

109 ), 

110 ) 

111 .withColumn( 

112 "feat_social_delta_pauvrete_5ans", 

113 F.round( 

114 ((F.col("feat_social_taux_pauvrete") - F.col("pauv17")) / F.col("pauv17")) * 100, 2 

115 ), 

116 ) 

117 .select( 

118 "id_arr_social", 

119 "feat_social_revenu_moyen", 

120 "feat_social_taux_pauvrete", 

121 "feat_social_pct_proprietaires", 

122 "feat_social_pct_logements_sociaux", 

123 "feat_social_delta_revenu_5ans", 

124 "feat_social_delta_pauvrete_5ans", 

125 ) 

126 ) 

127 

128 # Jointure Finale avec l'OBT 

129 logger.info("Fusion finale de toutes les dimensions dans l'OBT...") 

130 # On caste les deux clés en string pour être sûr 

131 df_obt = df_obt.withColumn("join_arr", F.col("code_insee_arrondissement").cast("string")) 

132 df_social_final = df_social_final.withColumn( 

133 "join_arr_social", F.col("id_arr_social").cast("string") 

134 ) 

135 

136 df_final = df_obt.join( 

137 df_social_final, df_obt.join_arr == df_social_final.join_arr_social, "left" 

138 ).drop("join_arr", "join_arr_social", "id_arr_social") 

139 

140 # Sauvegarde de la One Big Table (OBT) Finale 

141 output_path = os.path.join(GOLD_PATH, "ml", "obt_ml_complete") 

142 logger.info(f"Sauvegarde de l'OBT COMPLETE dans {output_path}...") 

143 df_final.write.mode("overwrite").parquet(output_path) 

144 

145 # Audit de fin 

146 logger.info(f"Pipeline terminée. Grain final : {df_final.count()} lignes (608 attendues).") 

147 logger.info(f"Nombre de features totales : {len(df_final.columns)}") 

148 df_final.show(5) 

149 

150 

151if __name__ == "__main__": 

152 run_gold_ml_step3_enrichment_social()
