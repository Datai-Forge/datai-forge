13SECURITE_SILVER_PATH = os.path.join(SILVER_PATH, "securite")

15def transform_bronze_to_silver(spark): 

16 """ 

17 Transforme les données de sécurité de la couche Bronze vers Silver. 

18 Application du filtrage Lyon et du typage des colonnes. 

19 """ 

20 bronze_full_path = os.path.join(SECURITE_BRONZE_PATH, "security-filtered") 

21 silver_full_path = os.path.join(SECURITE_SILVER_PATH, "lyon_securite") 

22 

23 if not os.path.exists(bronze_full_path): 

24 logger.error(f"Dossier Bronze introuvable : {bronze_full_path}") 

25 return 

26 

27 logger.info("Début de la transformation Silver pour la sécurité (Lyon)") 

28 

29 df = spark.read.parquet(bronze_full_path) 

30 

31 # 1. Sélection, Renommage et Conversion en Date (01/01/année) 

32 # 2. Casting des types numériques et gestion des estimations 

33 df_silver = df.select( 

34 to_date(concat(col("annee"), lit("-01-01")), "yyyy-MM-dd").alias("date_reference"), 

35 col("CODGEO_2025").alias("code_geo"), 

36 col("indicateur"), 

37 col("unite_de_compte"), 

38 # Règle Métier : Si le nombre est NULL (ndiff), on prend l'estimation du complément 

39 coalesce(col("nombre").cast(IntegerType()), col("complement_info_nombre").cast(IntegerType())).alias("nombre"), 

40 # Règle Métier : Idem pour le taux (redressement statistique) 

41 coalesce(col("taux_pour_mille").cast(DoubleType()), col("complement_info_taux").cast(DoubleType())).alias("taux_pour_1000"), 

42 col("insee_pop").cast(IntegerType()).alias("population"), 

43 col("insee_log").cast(IntegerType()).alias("logements"), 

44 # On garde quand même les colonnes d'estimation pour le ML (sans NULL) 

45 coalesce(col("complement_info_nombre").cast(DoubleType()), lit(0.0)).alias("nombre_estime"), 

46 coalesce(col("complement_info_taux").cast(DoubleType()), lit(0.0)).alias("taux_estime"), 

47 col("source_file"), 

48 col("bronze_processing_timestamp") 

49 ).withColumn("silver_processing_timestamp", current_timestamp()) 

50 

51 # 3. Filtrage : Uniquement Lyon ET Plage temporelle complète (2017 à 2022 inclus) 

52 df_silver = df_silver.filter( 

53 (col("code_geo").rlike("^6938[1-9]$")) & 

54 (year(col("date_reference")).between(2017, 2022)) 

55 ) 

56 

57 # Sauvegarde 

58 if not os.path.exists(SECURITE_SILVER_PATH): 

59 os.makedirs(SECURITE_SILVER_PATH, exist_ok=True) 

60 

61 df_silver.write.mode("overwrite").parquet(silver_full_path) 

62 

63 logger.info(f"Transformation Silver terminée : {silver_full_path}") 

64 logger.info(f"Nombre de lignes finales : {df_silver.count()}") 

65 df_silver.printSchema() 

66 

67if __name__ == "__main__": 

68 spark = get_spark_session(app_name="Silver_Securite_Lyon") 

69 

70 try: 

71 transform_bronze_to_silver(spark) 

72 finally: 

73 spark.stop() 

74 logger.info("Pipeline Silver Securite terminée.")
