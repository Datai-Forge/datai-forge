1import logging 

2import os 

3 

4from pyspark.sql import functions as F 

5 

6from src.common.spark_session_manager import get_spark_session 

7from src.config import GOLD_PATH, SILVER_PATH 

8 

9# Configuration du logging 

10logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s") 

11logger = logging.getLogger(__name__)

14def run_gold_bi_pipeline(): 

15 """ 

16 Transforme les données Silver en schéma en étoile pour la BI (Lyon). 

17 - dim_candidats : Référentiel des candidats 

18 - dim_geographie : Référentiel des bureaux de vote (avec type de bureau) 

19 - fact_votes : Nombre de voix par candidat et par tour 

20 - fact_participation : Inscrits, abstentions, votants, exprimés par tour 

21 """ 

22 spark = get_spark_session("Gold_Presidentielle_BI") 

23 logger.info("Démarrage de la pipeline Gold BI...") 

24 

25 # Chargement des données qui viennent de silver 

26 silver_t1_path = os.path.join(SILVER_PATH, "presidentielle", "lyon_T1_presidentiel_2022") 

27 silver_t2_path = os.path.join(SILVER_PATH, "presidentielle", "lyon_T2_presidentiel_2022") 

28 

29 df_t1 = spark.read.parquet(silver_t1_path).withColumn("tour", F.lit(1)) 

30 df_t2 = spark.read.parquet(silver_t2_path).withColumn("tour", F.lit(2)) 

31 

32 # Union des deux tours 

33 df_silver = df_t1.unionByName(df_t2) 

34 

35 # Dimension candidat 

36 logger.info("Extraction de dim_candidats...") 

37 dim_candidats = ( 

38 df_silver.select( 

39 "id_candidat", 

40 "nom", 

41 "prenom", 

42 "sexe", 

43 "parti_code", 

44 "parti_nom", 

45 "nuance_officielle", 

46 "bloc_analytique", 

47 ) 

48 .distinct() 

49 .filter(F.col("id_candidat").isNotNull()) 

50 ) 

51 

52 # Dimension géographique 

53 logger.info("Extraction de dim_geographie...") 

54 # Règle métier : Identifier le bureau 0001 comme Rattachement Administratif 

55 dim_geographie = ( 

56 df_silver.select( 

57 F.col("code_du_b_vote").alias("id_bureau"), "libelle_de_la_commune", "arrondissement" 

58 ) 

59 .distinct() 

60 .withColumn("code_insee", F.lit("69123")) 

61 .withColumn( 

62 "type_bureau", 

63 F.when(F.col("id_bureau") == "0001", "Rattachement Administratif").otherwise( 

64 "Standard" 

65 ), 

66 ) 

67 ) 

68 

69 # Table de faits pour les votes 

70 logger.info("Extraction de fact_votes...") 

71 fact_votes = df_silver.select( 

72 F.col("code_du_b_vote").alias("id_bureau"), "id_candidat", "tour", "voix" 

73 ).filter(F.col("id_candidat").isNotNull()) 

74 

75 # Table de faits pour la participation 

76 logger.info("Extraction de fact_participation...") 

77 # On prend une seule ligne par bureau et par tour 

78 fact_participation = df_silver.select( 

79 F.col("code_du_b_vote").alias("id_bureau"), 

80 "tour", 

81 "inscrits", 

82 "abstentions", 

83 "votants", 

84 "exprimes", 

85 ).distinct() 

86 

87 # Ajout d'indicateurs pré-calculés 

88 fact_participation = fact_participation.withColumn( 

89 "taux_participation", F.round((F.col("votants") / F.col("inscrits")) * 100, 2) 

90 ).withColumn("taux_abstention", F.round((F.col("abstentions") / F.col("inscrits")) * 100, 2)) 

91 

92 # pour la sauvegarde 

93 output_base_path = os.path.join(GOLD_PATH, "presidentielle", "bi") 

94 logger.info(f"Sauvegarde des tables Gold BI dans {output_base_path}...") 

95 

96 # On utilise des sous-dossiers explicites 

97 dim_candidats.write.mode("overwrite").parquet(os.path.join(output_base_path, "dim_candidats")) 

98 dim_geographie.write.mode("overwrite").parquet(os.path.join(output_base_path, "dim_geographie")) 

99 fact_votes.write.mode("overwrite").parquet(os.path.join(output_base_path, "fact_votes")) 

100 fact_participation.write.mode("overwrite").parquet( 

101 os.path.join(output_base_path, "fact_participation") 

102 ) 

103 

104 # Audit rapide 

105 total_voix = fact_votes.agg(F.sum("voix")).collect()[0][0] 

106 total_exprimes = fact_participation.agg(F.sum("exprimes")).collect()[0][0] 

107 

108 logger.info(f"Audit Gold BI : Total Voix={total_voix}, Total Exprimés={total_exprimes}") 

109 if total_voix == total_exprimes: 

110 logger.info("✅ Intégrité des données confirmée.") 

111 else: 

112 logger.warning("⚠️ Écart détecté entre les voix et les exprimés !") 

113 

114 logger.info("Pipeline Gold BI terminée avec succès !") 

115 

116 

117if __name__ == "__main__": 

118 run_gold_bi_pipeline()
